
from tkinter import *
from tkinter import messagebox
import base64
import os


def encrypt():
    password = code.get()
    message = text1.get("1.0", END).strip()

    if password == "":
        messagebox.showerror("Encryption", "Input Password")
    elif password != "1234":
        messagebox.showerror("Encryption", "Invalid Password")
    elif message == "":
        messagebox.showerror("Encryption", "Input Text")
    else:
        encode_message = message.encode("utf-8")
        base64_bytes = base64.b64encode(encode_message)
        encrypted_message = base64_bytes.decode("utf-8")

        text1.delete("1.0", END)
        text1.insert(END, encrypted_message)


def decrypt():
    password = code.get()
    message = text1.get("1.0", END).strip()

    if password == "":
        messagebox.showerror("Decryption", "Input Password")
    elif password != "1234":
        messagebox.showerror("Decryption", "Invalid Password")
    elif message == "":
        messagebox.showerror("Decryption", "Input Text")
    else:
        try:
            decode_message = message.encode("utf-8")
            base64_bytes = base64.b64decode(
                decode_message, validate=True
            )
            decrypted_message = base64_bytes.decode("utf-8")

            screen2 = Toplevel(screen)
            screen2.title("Decryption")
            screen2.geometry("400x200")
            screen2.configure(bg="#00bd56")

            Label(
                screen2,
                text="DECRYPT",
                font=("Arial", 14, "bold"),
                fg="white",
                bg="#00bd56"
            ).place(x=10, y=5)

            text2 = Text(
                screen2,
                font=("Roboto", 10),
                bg="white",
                relief=GROOVE,
                wrap=WORD,
                bd=0
            )
            text2.place(x=10, y=40, width=380, height=150)
            text2.insert(END, decrypted_message)

        except (ValueError, UnicodeDecodeError):
            messagebox.showerror(
                "Decryption", "Invalid encrypted text"
            )


def main_screen():
    global screen
    global code
    global text1

    screen = Tk()
    screen.geometry("400x370")
    screen.title("PctApp")
    screen.configure(bg="white")

    try:
        image_icon = PhotoImage(file="keys.png")
        screen.iconphoto(False, image_icon)
    except Exception:
        pass

    def reset():
        code.set("")
        text1.delete("1.0", END)

    Label(
        screen,
        text="Enter text for encryption and decryption",
        fg="black",
        bg="white",
        font=("Calibri", 13)
    ).place(x=10, y=10)

    text1 = Text(
        screen,
        font=("Roboto", 12),
        bg="white",
        relief=GROOVE,
        wrap=WORD,
        bd=1
    )
    text1.place(x=10, y=50, width=380, height=100)

    Label(
        screen,
        text="Enter Secret key for encryption and decryption",
        fg="black",
        bg="white",
        font=("Calibri", 12)
    ).place(x=10, y=165)

    code = StringVar()

    Entry(
        screen,
        textvariable=code,
        width=19,
        bd=1,
        font=("Arial", 18),
        show="*"
    ).place(x=10, y=195)

    # ENCRYPT button - left side
    Button(
        screen,
        text="ENCRYPT",
        height=2,
        width=16,
        bg="#ed3833",
        fg="white",
        bd=0,
        command=encrypt
    ).place(x=10, y=245)

    # DECRYPT button - right side
    Button(
        screen,
        text="DECRYPT",
        height=2,
        width=16,
        bg="#00bd56",
        fg="white",
        bd=0,
        command=decrypt
    ).place(x=210, y=245)

    # RESET button - bottom
    Button(
        screen,
        text="RESET",
        height=2,
        width=38,
        bg="#1089ff",
        fg="white",
        bd=0,
        command=reset
    ).place(x=10, y=300)

    screen.mainloop()


main_screen()
